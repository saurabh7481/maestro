import { randomUUID } from 'crypto';
import { Injectable } from '@nestjs/common';
 
@Injectable()
export class PaymentService {
  constructor(private gateway: Gateway) {}
 
  async charge(amount: number, card: Card, idempotencyKey?: string) {
    const key = idempotencyKey ?? randomUUID();
    return this.withRetry(() => this.gateway.charge(amount, card, key));
  }
}

my-app

A payments platform with pluggable gateways, idempotent charges, and automatic retry with backoff.

Getting started

Install dependencies and start the dev server:

$ pnpm install
$ pnpm dev

Features

  • Idempotency keys with a 24h TTL store
  • Exponential backoff, jittered, honoring Retry-After
  • Gateway adapters for Stripe, Adyen, and Braintree
Note: set GATEWAY_KEY in your .env before running.
